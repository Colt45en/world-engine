Code Examples

C++ Example


#include <world_engine/world_engine.h>

int main() {
    WorldEngine::Engine engine;
    engine.initialize();
    
    auto scene = std::make_shared<WorldEngine::Scene>("Main");
    auto entity = scene->createEntity("Player");
    
    engine.setActiveScene(scene);
    engine.run();
    
    return 0;
}

Python Example


import world_engine as we

engine = we.Engine()
engine.initialize()

scene = we.Scene("Main")
entity = scene.create_entity("Player")

engine.set_active_scene(scene)
engine.run()

TypeScript Example


import { Engine, Scene } from '@worldengine/core';

const engine = new Engine();
engine.initialize();

const scene = new Scene('Main');
const entity = scene.createEntity('Player');

engine.setActiveScene(scene);
// In browser/Node.js, use tick() for frame updates
setInterval(() => engine.tick(), 16);
